# End-to-end A02 repair journey on a copy of fix_real (real chart, branches,
# stock), driven through the modules' own actions. Superuser: the role and SoD
# gates are covered by the unit suites; this proves the business chain and the
# ledger across M04 -> M01 -> M31 -> M15 -> M04 drawer -> M20 -> M51 -> M19 ->
# M50 -> M09/M27. Every step is reported; the walk commits nothing.
import base64, traceback
from odoo import fields
from odoo.addons.cd_service_order.models import fix_otp_util

env = env(user=env.ref("base.user_admin").id, su=True)
P = env["ir.config_parameter"].sudo()
sent = {}
fix_otp_util.dispatch_otp = lambda phone, code, ttl, language="bn": sent.update({phone: code})
results = []
ctx = {}


def step(name):
    def deco(fn):
        try:
            with env.cr.savepoint():
                out = fn()
            results.append(("OK  ", name, out))
            return out
        except Exception as e:  # noqa: BLE001
            results.append(("FAIL", name, "%s: %s" % (type(e).__name__, str(e)[:300])))
            ctx["tb_" + name] = traceback.format_exc()[-1500:]
            return None
    return deco


def acc(code):
    return env["account.account"].search([("code", "=", code)], limit=1)


def bal(account, partner=None):
    env.flush_all()
    dom = [("account_id", "=", account.id), ("parent_state", "=", "posted")]
    if partner:
        dom.append(("partner_id", "=", partner.id))
    return round(sum(env["account.move.line"].search(dom).mapped("balance")), 2)


company = env.company
branch = env["fix.branch"].search(
    [("operating_unit_id", "!=", False), ("name", "ilike", "Shewrapara")], limit=1
)
drawer = acc(P.get_param("cd_cash_position.account_code.branch_cash"))
ar = acc(P.get_param("cd_billing_engine.account_code.ar"))
rev = acc(P.get_param("cd_billing_engine.account_code.revenue"))
vat = acc(P.get_param("cd_billing_engine.account_code.vat_output"))
d0, rev0, vat0 = bal(drawer), bal(rev), bal(vat)
cust = env["res.partner"].create({"name": "E2E Customer Rahim", "phone": "01711999888"})
brand = env["fix.brand"].search([], limit=1)
model = env["fix.device.model"].search([("brand_id", "=", brand.id)], limit=1) or env[
    "fix.device.model"
].create({"brand_id": brand.id, "name": "E2E Laptop 15", "device_type": "laptop"})
emp = env["hr.employee"].search([("company_id", "=", company.id)], limit=1)
tech = env["hr.employee"].search([("company_id", "=", company.id), ("user_id", "!=", False)], limit=1)

mushak0 = env["fix.mushak.register"].search_count([])


@step("M04 open the branch cash shift")
def _():
    Shift = env["fix.cash.shift"]
    s = Shift._fd_open_shift_for(branch, company) or Shift.create(
        {"branch_id": branch.id, "company_id": company.id,
         "operating_unit_id": branch.operating_unit_id.id, "operator_id": emp.id,
         "opening_float": 0.0})
    ctx["shift"] = s
    return "shift #%s %s" % (s.id, s.state)


@step("M17 register the customer's device")
def _():
    dev = env["fix.device"].create(
        {"device_model_id": model.id, "owner_id": cust.id, "serial_number": "E2E-SN-DECL"})
    ctx["device"] = dev
    return dev.display_name


@step("M04 walk-in intake with a BDT 300 cash diagnostic fee, hand-over")
def _():
    it = env["fix.walkin.intake"].create(
        {"branch_id": branch.id, "operating_unit_id": branch.operating_unit_id.id,
         "customer_id": cust.id, "device_id": ctx["device"].id,
         "fault_description": "No display after drop", "diagnostic_fee_bdt": 300.0,
         "payment_channel": "cash"})
    sig = env["ir.attachment"].create({"name": "signed-intake.png", "raw": b"signed by Rahim",
                                       "res_model": it._name, "res_id": it.id})
    it.write({"handover_method": "signature", "customer_signature_attachment_id": sig.id,
              "accessory_bag_barcode": "BAG-E2E-DECL"})
    it.action_complete_handover()
    ctx["order"] = it.service_order_id
    ctx["fee"] = r = it.diagnostic_fee_receipt_id
    return "order %s, fee receipt %s %s Dr %s, unapplied %.2f" % (
        it.service_order_id.name, r.name, r.state, r.account_id.code, r.unapplied_amount)


@step("M01 diagnose and send the estimate (part + labour)")
def _():
    o = ctx["order"]
    o.action_start_diagnostic()
    part = env["product.product"].search([("default_code", "!=", False), ("type", "!=", "service")], limit=1)
    est = env["fix.estimate"].create({"order_id": o.id, "line_ids": [
        (0, 0, {"description": part.name or "Display", "product_id": part.id, "quantity": 1,
                "unit_price": 6000.0, "line_type": "part"}),
        (0, 0, {"description": "Display replacement labour", "quantity": 1,
                "unit_price": 1000.0, "line_type": "labour"})]})
    o.action_request_estimate() if o.state == "diagnostic" else None
    est.action_send()
    o.invalidate_recordset()
    return "estimate %s total %.2f, order %s" % (est.state, est.grand_total, o.state)


@step("M01 the customer rejects the estimate -> order cancelled (estimate_declined)")
def _():
    o = ctx["order"]
    o.action_cancel(reason_code="estimate_declined", note="Too expensive, buying a new one")
    o.invalidate_recordset()
    inv = env["fix.ar.invoice"].search([("source_model", "=", o._name), ("source_id", "=", o.id)])
    ctx["invoice"] = inv
    line = inv.line_ids[:1]
    return "order %s; fee invoice %s %s total %.2f (taxable %.2f + VAT %.2f) residual %.2f; line %s %s" % (
        o.state, inv.name, inv.state, inv.total_payable, inv.taxable_amount, inv.vat_amount,
        inv.amount_residual, line.line_type, line.product_id.default_code)


@step("M51 the fee carries its Mushak 6.3")
def _():
    inv = ctx["invoice"]
    assert inv.mushak_id, "no Mushak row"
    return "Mushak %s (register rows +%d), GL move %s" % (
        inv.mushak_id.mushak_no, env["fix.mushak.register"].search_count([]) - mushak0,
        inv.move_id.name if "move_id" in inv._fields and inv.move_id else "-")


@step("M20 the fee receipt is applied, not left as credit")
def _():
    r = ctx["fee"]
    r.invalidate_recordset()
    assert abs(r.unapplied_amount) < 0.005, r.unapplied_amount
    return "receipt %s unapplied %.2f" % (r.name, r.unapplied_amount)


@step("M19 ledger after the decline")
def _():
    d = bal(drawer) - d0
    a = bal(ar, cust)
    rv = bal(rev) - rev0
    vt = bal(vat) - vat0
    assert abs(a) < 0.005 and abs(rv + vt + 300.0) < 0.005, (a, rv, vt)
    return "drawer +%.2f, customer AR %.2f, revenue %+.2f, VAT %+.2f" % (d, a, rv, vt)


@step("M01 the order chatter tells the counter what happened")
def _():
    body = " ".join(ctx["order"].message_ids.mapped("body"))
    assert "diagnostic fee invoiced" in body, body[:300]
    return "chatter ok"


for flag, name, out in results:
    print(flag, name, "->", out)
for k, v in ctx.items():
    if k.startswith("tb_"):
        print("\n" + k, v)
print("SUMMARY", sum(1 for r in results if r[0] == "OK  "), "/", len(results))
env.cr.rollback()
