# End-to-end A02 repair journey on a copy of fix_real (real chart, branches,
# stock), driven through the modules' own actions. Superuser: the role and SoD
# gates are covered by the unit suites; this proves the business chain and the
# ledger across M04 -> M01 -> M31 -> M15 -> M04 drawer -> M20 -> M51 -> M19 ->
# M50 -> M09/M27. Every step is reported; the walk commits nothing.
import base64, traceback
from odoo import fields
from odoo.addons.cd_service_order.models import fix_otp_util

env = env(user=env.ref("base.user_admin").id, su=True)
P = env["ir.config_parameter"].sudo()
sent = {}
fix_otp_util.dispatch_otp = lambda phone, code, ttl, language="bn": sent.update({phone: code})
results = []
ctx = {}


def step(name):
    def deco(fn):
        try:
            with env.cr.savepoint():
                out = fn()
            results.append(("OK  ", name, out))
            return out
        except Exception as e:  # noqa: BLE001
            results.append(("FAIL", name, "%s: %s" % (type(e).__name__, str(e)[:300])))
            ctx["tb_" + name] = traceback.format_exc()[-1500:]
            return None
    return deco


def acc(code):
    return env["account.account"].search([("code", "=", code)], limit=1)


def bal(account, partner=None):
    env.flush_all()
    dom = [("account_id", "=", account.id), ("parent_state", "=", "posted")]
    if partner:
        dom.append(("partner_id", "=", partner.id))
    return round(sum(env["account.move.line"].search(dom).mapped("balance")), 2)


company = env.company
branch = env["fix.branch"].search(
    [("operating_unit_id", "!=", False), ("name", "ilike", "Shewrapara")], limit=1
)
drawer = acc(P.get_param("cd_cash_position.account_code.branch_cash"))
ar = acc(P.get_param("cd_billing_engine.account_code.ar"))
rev = acc(P.get_param("cd_billing_engine.account_code.revenue"))
vat = acc(P.get_param("cd_billing_engine.account_code.vat_output"))
d0, rev0, vat0 = bal(drawer), bal(rev), bal(vat)
cust = env["res.partner"].create({"name": "E2E Customer Rahim", "phone": "01711999888"})
brand = env["fix.brand"].search([], limit=1)
model = env["fix.device.model"].search([("brand_id", "=", brand.id)], limit=1) or env[
    "fix.device.model"
].create({"brand_id": brand.id, "name": "E2E Laptop 15", "device_type": "laptop"})
emp = env["hr.employee"].search([("company_id", "=", company.id)], limit=1)
tech = env["hr.employee"].search([("company_id", "=", company.id), ("user_id", "!=", False)], limit=1)
for comp in env["fix.competency"].search([("required_for_dispatch", "=", True)]):
    cert = env["fix.certification"].create({"employee_id": tech.id, "competency_id": comp.id,
        "brand_id": comp.brand_id.id or False, "repair_tier": "l3",
        "issued_on": fields.Date.today(), "expires_on": fields.Date.add(fields.Date.today(), years=1)})
    if cert.state != "valid":
        (cert._guarded() if hasattr(cert, "_guarded") else cert).write({"state": "valid"})

@step("M01 diagnose and estimate (part + labour), customer approves")
def _():
    o = ctx["order"]
    o.action_start_diagnostic()
    part = env["product.product"].search([("default_code", "!=", False), ("type", "!=", "service")], limit=1)
    est = env["fix.estimate"].create({"order_id": o.id, "line_ids": [
        (0, 0, {"description": part.name or "Display", "product_id": part.id, "quantity": 1,
                "unit_price": 2000.0, "line_type": "part"}),
        (0, 0, {"description": "Display replacement labour", "quantity": 1,
                "unit_price": 1000.0, "line_type": "labour"})]})
    o.action_request_estimate() if o.state == "diagnostic" else None
    est.action_send()
    est.action_approve()

@step("M31 book into the workshop, repair, QC pass -> order Ready")
def _():
    o = ctx["order"]
    if o.state == "approved":
        o.action_start_repair()
    o.action_submit_qc()
    chk = env["fix.qc.checklist"].create({"order_id": o.id, "item_ids": [
        (0, 0, {"name": n, "passed": True}) for n in ("Power on", "Display", "Charging")]})
    if hasattr(chk, "action_complete"):
        chk.action_complete()
    o.action_ready()

@step("M19 ledger after delivery")
def _():
    return "drawer +%.2f, customer AR %.2f, revenue %+.2f, VAT %+.2f" % (
        bal(drawer) - d0, bal(ar, cust), bal(rev) - rev0, bal(vat) - vat0)


@step("M01 close -> M27/M34 events")
def _():
    o = ctx["order"]
    o.action_close()
    ev = env["fix.event.outbox"].search([("aggregate", "=", o._name), ("aggregate_id", "=", o.id)]).mapped("event_type")
    return "order %s; events: %s" % (o.state, ", ".join(sorted(set(ev))))


@step("M04 count and close the drawer; M50 deposit dispatch")
def _():
    s = ctx["shift"]
    s.action_close_shift()
    rest = round(s.expected_closing, 2)
    notes = (("denom_1000_count", 1000), ("denom_500_count", 500), ("denom_200_count", 200),
             ("denom_100_count", 100), ("denom_50_count", 50), ("denom_20_count", 20), ("denom_10_count", 10))
    count = {}
    for n, v in notes:
        count[n], rest = int(rest // v), round(rest % v, 2)
    count["coins_amount"] = rest
    s.write(count)
    s.action_submit_blind_count()
    s.action_confirm_closed()
    bank = env["account.journal"].search([("type", "=", "bank"), ("company_id", "=", company.id)], limit=1)
    Dep = env["fix.cash.deposit"]
    dep = Dep.create({"operating_unit_id": branch.operating_unit_id.id, "branch_id": branch.id,
                      "journal_id": bank.id, "business_date": s.shift_date,
                      "amount": s.closing_counted - s.opening_float,
                      "carrier": Dep._fields["carrier"].selection[0][0], "slip_no": "E2E-SLIP-1"})
    dep.action_dispatch()
    return "shift %s expected %.2f counted %.2f; deposit %.2f; drawer net %+.2f" % (
        s.state, s.expected_closing, s.closing_counted, dep.amount, bal(drawer) - d0)

@step("S2 M31/M15 issue a stocked part to the job at its stock cost")
def _():
    job = ctx["job"]
    onhand = env["fix.stock.onhand"].search([("qty", ">", 0), ("unit_cost", ">", 0)], limit=1)
    part = onhand.part_id
    # setup the walk supplies (fix_real has no workshop<->store link): the
    # workshop draws from the store that holds the part
    if not job.workshop_id._a06_m15_store():
        job.workshop_id.sudo().write({"m15_location_id": onhand.location_id.id})
    before = onhand.qty
    PL = env["fix.workshop.part.line"]
    vals = {"job_id": job.id, "quantity": 1}
    for f, v in (("spare_part_id", part.id), ("part_id", part.id),
                 ("product_id", part.product_id.id if part.product_id else False),
                 ("location_id", onhand.location_id.id)):
        if f in PL._fields and v:
            vals[f] = v
    line = PL.create(vals)
    if "technician_id" in job._fields and not job.technician_id:
        (job._guarded() if hasattr(job, "_guarded") else job).write({"technician_id": tech.id})
    for st in ("action_intake", "action_diagnostic", "action_auto_assign_bench", "action_start_repair"):
        if job.state != "in_repair":
            try:
                with env.cr.savepoint():
                    getattr(job, st)()
            except Exception as e:  # noqa: BLE001
                ctx.setdefault("job_gates", []).append("%s: %s" % (st, str(e)[:220]))
    if job.state != "in_repair":
        raise RuntimeError("job is %s; gates: %s" % (job.state, " | ".join(ctx.get("job_gates", []))))
    if hasattr(line, "action_reserve"):
        line.action_reserve()
    line.action_consume_part()
    onhand.invalidate_recordset()
    job.invalidate_recordset()
    ctx["line"] = line
    return "part %s line cost %.2f (bucket %.2f), stock %s -> %s, job parts_total %.2f, job %s" % (
        part.part_number, line.unit_cost, onhand.unit_cost, before, onhand.qty, job.parts_total, job.state)


@step("S2 M31 QC on the bench -> job and order Ready")
def _():
    job, o = ctx["job"], ctx["o2"]
    # the release evidence the technician files on the bench
    Ev = env["fix.workshop.evidence"]
    for kind, title in (("post_repair_photo", "Finished device"), ("test_log", "Functional test")):
        Ev.create({"job_id": job.id, "name": title, "evidence_type": kind,
                   "upload_file": base64.b64encode(b"PNG evidence"), "upload_filename": kind + ".png"})
    job.action_submit_qc()
    if job.state == "burn_in" or (hasattr(job, "burn_in_required") and job.burn_in_required):
        job.action_start_burn_in()
        job.action_pass_burn_in()
    if job.state == "qc_pending":
        job.action_pass_qc(notes="All functions tested")
    job.invalidate_recordset()
    o.invalidate_recordset()
    return "job %s (labour cost %.2f, total cost %.2f), order %s" % (
        job.state, getattr(job, "labour_cost", 0.0), getattr(job, "total_repair_cost", 0.0), o.state)

    m3 = env["fix.device.model"].search([("brand_id", "=", lap.brand_id.id), ("device_type", "=", "laptop")], limit=1)         or env["fix.device.model"].create({"brand_id": lap.brand_id.id, "name": "E2E Brand Laptop", "device_type": "laptop"})
    dev = env["fix.device"].create({"device_model_id": m3.id, "owner_id": cust.id,
                                    "serial_number": "E2E-SN-0003"})
    o = env["fix.service.order"].create({"device_id": dev.id, "customer_id": cust.id,
        "branch_id": branch.id, "fault_description": "AC not cooling", "service_type": "onsite",
        "order_category": "standard"})
    o.action_intake()
    o.action_start_diagnostic()
    est = env["fix.estimate"].create({"order_id": o.id, "line_ids": [
        (0, 0, {"description": "Gas refill and service", "quantity": 1, "unit_price": 1500.0,
                "line_type": "labour"})]})
    if o.state == "diagnostic":
        o.action_request_estimate()
    est.action_send()
    est.action_approve()

    fo.action_request_dispatch()
    fo.action_dispatch(technician_id=tech.id)

@step("S3 M02 cash collected on site, technician declares, cashier accepts")

    fo.action_record_collection(amount=due, method="cash")

    fo.action_complete()
    fo.action_handover_field_cash()

         "operating_unit_id": branch.operating_unit_id.id, "operator_id": emp.id,
         "opening_float": 0.0})

    cashier = env["res.users"].with_context(no_reset_password=True).create({

    return "collected %.2f; field %s / %s; order settled %s verified %s; receipt %s Dr %s; drawer field cash %.2f" % (
